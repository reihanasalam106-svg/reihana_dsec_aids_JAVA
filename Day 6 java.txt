1) Insertion 

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc= new Scanner(System.in);
        int a[] = new int[6];
        a[0] = 10;
        a[1] = 20;
        a[2] = 30;
        a[3] = 40;
        a[4] = 50;
        int n = 5;
        System.out.println("Enter value:");
        int value = sc.nextInt();
        System.out.println("Enter value(0 to " + n +"):");
        int pos = sc.nextInt();
        if(pos<0 || pos >n || n==a.length){
            System.out.println("Invalid insertion");
        }else{
            for(int i=n; i>pos; i--){
                a[i] = a[i-1];            
            }
            a[pos]=value;
            n++;
            for(int i=0; i<n; i++){
                System.out.println(a[i]+" ");
            }
        }
        sc.close();
        
    }
}

4) Linear Search

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        int a[]={40,10,30,20,50};
        int key =30;
        int pos = -1;
        for(int i = 0; i<a.length; i++){
            if(a[i] == key){
                pos=1;
                break;
            }
        }
        if(pos!=-1)
            System.out.println("Found at index"+ pos);
        else
            System.out.println("Not Found");
    }
}           

5) Binary Search

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        int a[]={10,20,30,40,50,60};
        int key =40;
        int low = 0;
        int high =a.length - 1;
        int pos = -1;
        while(low<=high){
            int mid = (low + high)/2;
            if(a[mid]==key){
                pos = mid;
                break;
            }else if(key>a[mid]){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
        if(pos!=-1)
            System.out.println("Found at index"+ pos);
        else
            System.out.println("Not Found");
    }
}

8) Second Largest Element

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int a[]=new int [5];
        System.out.println("Enter 5 elements");
        for(int i =0; i<5; i++){
            a[i] = sc.nextInt();
        }
        int largest = a[0];
        int second = Integer.MIN_VALUE;
        for(int i = 1; i<5 ; i++){
            if(a[i]>largest){
                second = largest;
                largest = a[i];
            }else if(a[i]>second && a[i] < largest){
                second = a[i];
            }
        }
        if(second == Integer.MIN_VALUE){
            System.out.println("Second Largest element not found");
        }else{
            System.out.println("Second Largest =" + second);
        }
        
    }
} 

9) Array after moving zeros

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        int a[]={0,5,0,3,8,0,2,4,2,0,0,3,7,0,2,0};
        int j=0;
        for(int i=0;i<a.length;i++){
            if(a[i]!=0){
                a[j]=a[i];
                j++;
            }
        }
        while(j<a.length){
            a[j]=0;
            j++;
        }
        System.out.println("Array after moving zeros:");
        for(int i=0;i<a.length;i++){
            System.out.println(a[i]+" ");
        }
        
    }
} 

10) Unique Values

class Main {
    public static void main(String[] args) {
        int a[] = {1,1,5,3,7,5};
        System.out.println("Unique values :");
        for(int i=0; i<a.length;i++){
            int count =0;
            for(int j=0;j<i;j++){
                if(a[i]==a[j]){
                    count++;
                    break;
                }
            }
            if(count==0){
                System.out.println(a[i] + " ");
            }
        }
    }
}
