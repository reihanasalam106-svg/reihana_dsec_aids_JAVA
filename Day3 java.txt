15) Number Relationship

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();

        if (a == 0 || b == 0) {
            System.out.println("Cannot Perform Divisibility Check");
        } else {
            if (a > b) {
                if (a % b == 0) {
                    System.out.println("Greater Number is a Multiple of Smaller Number");
                } else {
                    System.out.println("Greater Number is Not a Multiple");
                }
            } else if (b > a) {
                if (b % a == 0) {
                    System.out.println("Greater Number is a Multiple of Smaller Number");
                } else {
                    System.out.println("Greater Number is Not a Multiple");
                }
            } else {
                System.out.println("Both Numbers are Equal");
            }
        }
    }
}

16) Calculator Menu


import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double a = sc.nextDouble();
        double b = sc.nextDouble();
        char op = sc.next().charAt(0);

        switch (op) {
            case '+':
                System.out.println("Result: " + (a + b));
                break;
            case '-':
                System.out.println("Result: " + (a - b));
                break;
            case '*':
                System.out.println("Result: " + (a * b));
                break;
            case '/':
                if (b == 0) {
                    System.out.println("Cannot Divide by Zero");
                } else {
                    System.out.println("Result: " + (a / b));
                }
                break;
            default:
                System.out.println("Invalid Operator");
        }
    }
}

18) Food Order Menu


import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int choice = sc.nextInt();
        int quantity = sc.nextInt();
        int price;
        String food;

        if (quantity <= 0) {
            System.out.println("Invalid Quantity");
        } else {
            switch (choice) {
                case 1:
                    food = "Burger";
                    price = 120;
                    break;
                case 2:
                    food = "Pizza";
                    price = 250;
                    break;
                case 3:
                    food = "Sandwich";
                    price = 100;
                    break;
                case 4:
                    food = "Pasta";
                    price = 180;
                    break;
                default:
                    System.out.println("Invalid Choice");
                    sc.close();
                    return;
            }

            System.out.println("Food: " + food);
            System.out.println("Total: Rs. " + (price * quantity));
        }

        sc.close();
    }
}

20) Mini Banking Menu


import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double balance = sc.nextDouble();
        int choice = sc.nextInt();

        switch (choice) {
            case 1:
                System.out.println("Balance: Rs. " + balance);
                break;

            case 2:
                double deposit = sc.nextDouble();

                if (deposit > 0) {
                    balance = balance + deposit;
                    System.out.println("Deposit Successful");
                    System.out.println("Updated Balance: Rs. " + balance);
                } else {
                    System.out.println("Invalid Amount");
                }
                break;

            case 3:
                double withdraw = sc.nextDouble();

                if (withdraw > 0) {
                    if (withdraw <= balance) {
                        balance = balance - withdraw;
                        System.out.println("Withdrawal Successful");
                        System.out.println("Remaining Balance: Rs. " + balance);
                    } else {
                        System.out.println("Insufficient Balance");
                    }
                } else {
                    System.out.println("Invalid Amount");
                }
                break;

            case 4:
                System.out.println("Exit");
                break;

            default:
                System.out.println("Invalid Choice");
        }
    }
}
